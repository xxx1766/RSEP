package routine
import "testing"
func TestAuditCPUOnlyFilter(t *testing.T) {
 remaining := &Resource{CpuNum: 6, GpuNum: 0}
 request := &Resource{CpuNum: 2, GpuNum: 0}
 if remaining.Greater(EmptyResourse) { t.Fatal("unexpected: CPU-only candidate passed current scheduler predicate") }
 if !remaining.GreaterOrEqual(request) { t.Fatal("request should fit capacity") }
 t.Log("CONFIRMED: CPU-only node has enough capacity but current Greater(empty) candidate filter excludes it")
}
func TestAuditIncomparableCapacity(t *testing.T) {
 remaining := &Resource{CpuNum: 4, GpuNum: 1}
 request := &Resource{CpuNum: 2, GpuNum: 2}
 t.Logf("available<=request=%v; request<=available=%v", request.GreaterOrEqual(remaining), remaining.GreaterOrEqual(request))
 if request.GreaterOrEqual(remaining) || remaining.GreaterOrEqual(request) { t.Fatal("expected incomparable resource vectors") }
}
